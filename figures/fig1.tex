\newcommand{\cpanel}[9]{%

    \node[anchor=west, font=\bfseries\small] at (0,7.9) {#2};
    \node[anchor=west, font=\small] at (0.45,7.9) {#3};
    \foreach \i in {0,1,2}{
      \node[tokfixed] at (\i*1.05+0.6,7.1) {};}
    \foreach \i in {3,4}{

    \node[font=\scriptsize, anchor=north] at (4.28,6.62) {#5};
    \node[fixedbox] (mod) at (2.65,5.55) {#6};
    \node[varbox]  (obj) at (2.65,4.15) {\shortstack{#7\\[1pt]\scriptsize #8}};
    \node[varbox]  (grd) at (2.65,2.75) {\shortstack{Gradient\\[1pt]\scriptsize #9}};
    \node[varbox]  (cnd) at (2.65,1.35) {\shortstack{Candidate batch\\[1pt]\scriptsize top rows of the table}};

    \draw[flow] (grd) -- (cnd);
